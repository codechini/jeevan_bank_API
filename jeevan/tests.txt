# Quick registration and login test:
# 1. Register
RESPONSE=$(curl -s -v -X POST http://localhost:8080/api/auth/register \
  -H "Content-Type: application/json" \
  -d '{"username":"quicktest","password":"QuickTest123!","email":"quicktest@test.com","firstName":"Quick","lastName":"Test"}')
echo "Register Response: $RESPONSE"

# 2. Extract token (requires jq)
TOKEN=$(echo $RESPONSE | jq -r '.data.token')
echo "Token: $TOKEN"

# 3. Open account using the token
curl -v -X POST http://localhost:8080/api/user/openaccount \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer $TOKEN" \
  -d '{"firstName":"Quick","lastName":"Test","accountType":"SAVINGS"}'
