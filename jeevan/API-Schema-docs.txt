### 4\. Security Configuration (JWT)

**JwtUtils.java** (Handles token generation)

```java
package com.jeevanbank.security;

import io.jsonwebtoken.*;
import io.jsonwebtoken.security.Keys;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.security.core.Authentication;
import org.springframework.security.core.userdetails.UserDetails;
import org.springframework.stereotype.Component;
import java.security.Key;
import java.util.Date;

@Component
public class JwtUtils {
    @Value("${app.jwtSecret}")
    private String jwtSecret;

    @Value("${app.jwtExpirationMs}")
    private int jwtExpirationMs;

    public String generateJwtToken(Authentication authentication) {
        UserDetails userPrincipal = (UserDetails) authentication.getPrincipal();
        return Jwts.builder()
                .setSubject((userPrincipal.getUsername()))
                .setIssuedAt(new Date())
                .setExpiration(new Date((new Date()).getTime() + jwtExpirationMs))
                .signWith(key(), SignatureAlgorithm.HS256)
                .compact();
    }

    private Key key() {
        return Keys.hmacShaKeyFor(jwtSecret.getBytes());
    }

    public String getUserNameFromJwtToken(String token) {
        return Jwts.parserBuilder().setSigningKey(key()).build()
               .parseClaimsJws(token).getBody().getSubject();
    }

    public boolean validateJwtToken(String authToken) {
        try {
            Jwts.parserBuilder().setSigningKey(key()).build().parseClaimsJws(authToken);
            return true;
        } catch (JwtException e) {
            System.err.println("Invalid JWT: " + e.getMessage());
        }
        return false;
    }
}
```

### 3\. The Transaction Service (Core Logic)

This service performs the logic defined in your `Transaction` and `Account` schema definitions.

**Key Logic Implemented:**

1.  **Atomic Transactions:** Uses `@Transactional` so if the database fails after deducting money but before adding it, the entire operation rolls back.
2.  **Double-Entry Logging:** Creates two records in the `Transaction` table—one `DEBIT` for the sender and one `CREDIT` for the receiver—so both users see the history in their own statements.
3.  **Balance Check:** Ensures the sender cannot overdraw.

<!-- end list -->

```java
package com.jeevanbank.service;

import com.jeevanbank.dto.TransferRequest;
import com.jeevanbank.entity.Account;
import com.jeevanbank.entity.Transaction;
import com.jeevanbank.repository.AccountRepository;
import com.jeevanbank.repository.TransactionRepository;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.math.BigDecimal;
import java.time.LocalDateTime;

@Service
public class AccountService {

    @Autowired
    private AccountRepository accountRepository;

    @Autowired
    private TransactionRepository transactionRepository;

    @Transactional // CRITICAL: Ensures all DB changes happen together or fail together
    public void transferFunds(TransferRequest request) {
        
        // 1. Validate Amount
        if (request.getAmount().compareTo(BigDecimal.ZERO) <= 0) {
            throw new RuntimeException("Transfer amount must be greater than zero");
        }

        // 2. Fetch Accounts
        Account sender = accountRepository.findByAccountNumber(request.getFromAccountNumber())
                .orElseThrow(() -> new RuntimeException("Sender account not found"));

        Account receiver = accountRepository.findByAccountNumber(request.getToAccountNumber())
                .orElseThrow(() -> new RuntimeException("Receiver account not found"));

        // 3. Check Sufficient Funds
        if (sender.getBalance().compareTo(request.getAmount()) < 0) {
            throw new RuntimeException("Insufficient balance");
        }

        // 4. Update Balances (In-Memory)
        sender.setBalance(sender.getBalance().subtract(request.getAmount()));
        receiver.setBalance(receiver.getBalance().add(request.getAmount()));

        // 5. Save Account Updates to DB
        accountRepository.save(sender);
        accountRepository.save(receiver);

        // 6. Log Transaction for Sender (DEBIT)
        Transaction debitTx = new Transaction();
        debitTx.setAccount(sender); // Maps to account_id UUID in schema
        debitTx.setTransactionType("DEBIT");
        debitTx.setAmount(request.getAmount().negate()); // Show as negative for sender
        debitTx.setDescription("Transfer to " + receiver.getAccountNumber() + ": " + request.getDescription());
        debitTx.setRelatedAccountNumber(receiver.getAccountNumber());
        debitTx.setStatus("Completed");
        debitTx.setTimestamp(LocalDateTime.now());
        
        // 7. Log Transaction for Receiver (CREDIT)
        Transaction creditTx = new Transaction();
        creditTx.setAccount(receiver); // Maps to account_id UUID in schema
        creditTx.setTransactionType("CREDIT");
        creditTx.setAmount(request.getAmount()); // Show as positive for receiver
        creditTx.setDescription("Transfer from " + sender.getAccountNumber() + ": " + request.getDescription());
        creditTx.setRelatedAccountNumber(sender.getAccountNumber());
        creditTx.setStatus("Completed");
        creditTx.setTimestamp(LocalDateTime.now());

        // 8. Save Transactions
        transactionRepository.save(debitTx);
        transactionRepository.save(creditTx);
    }
}
```

```java
package com.jeevanbank.service;

import com.jeevanbank.dto.LoginRequest;
import com.jeevanbank.dto.RegisterRequest;
import com.jeevanbank.dto.JwtResponse;
import com.jeevanbank.entity.Role;
import com.jeevanbank.entity.User;
import com.jeevanbank.repository.RoleRepository;
import com.jeevanbank.repository.UserRepository;
import com.jeevanbank.security.JwtUtils;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.security.authentication.AuthenticationManager;
import org.springframework.security.authentication.UsernamePasswordAuthenticationToken;
import org.springframework.security.core.Authentication;
import org.springframework.security.core.context.SecurityContextHolder;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.stereotype.Service;

@Service
public class AuthService {

    @Autowired
    AuthenticationManager authenticationManager;

    @Autowired
    UserRepository userRepository;

    @Autowired
    RoleRepository roleRepository;

    @Autowired
    PasswordEncoder encoder;

    @Autowired
    JwtUtils jwtUtils;

    public JwtResponse login(LoginRequest loginRequest) {
        Authentication authentication = authenticationManager.authenticate(
                new UsernamePasswordAuthenticationToken(loginRequest.getUsername(), loginRequest.getPassword()));

        SecurityContextHolder.getContext().setAuthentication(authentication);
        String jwt = jwtUtils.generateJwtToken(authentication);
        
        User userDetails = userRepository.findByUsername(loginRequest.getUsername()).get();
        
        return new JwtResponse(jwt, userDetails.getUserId(), userDetails.getUsername(), userDetails.getEmail(), userDetails.getRole().getRoleName());
    }

    public String register(RegisterRequest signUpRequest) {
        if (userRepository.existsByUsername(signUpRequest.getUsername())) {
            throw new RuntimeException("Error: Username is already taken!");
        }

        if (userRepository.existsByEmail(signUpRequest.getEmail())) {
            throw new RuntimeException("Error: Email is already in use!");
        }

        // Create new user's account
        User user = new User();
        user.setUsername(signUpRequest.getUsername());
        user.setEmail(signUpRequest.getEmail());
        user.setPassword(encoder.encode(signUpRequest.getPassword()));

        // Assign Default Role (User)
        Role userRole = roleRepository.findByRoleName("ROLE_USER")
                .orElseThrow(() -> new RuntimeException("Error: Role is not found."));
        user.setRole(userRole);

        userRepository.save(user);

        return "User registered successfully!";
    }
}
```

### 2\. Account Service (Extended)

```java
package com.jeevanbank.service;

import com.jeevanbank.dto.OpenAccountRequest;
import com.jeevanbank.dto.AmountRequest;
import com.jeevanbank.entity.*;
import com.jeevanbank.repository.*;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.math.BigDecimal;
import java.time.LocalDateTime;
import java.util.List;
import java.util.Random;
import java.util.UUID;

@Service
public class AccountService {

    @Autowired
    private AccountRepository accountRepository;
    @Autowired
    private AccountHolderRepository holderRepository;
    @Autowired
    private UserRepository userRepository;
    @Autowired
    private TransactionRepository transactionRepository;

    // --- OPEN ACCOUNT LOGIC ---
    @Transactional
    public Account openAccount(UUID userId, OpenAccountRequest request) {
        User user = userRepository.findById(userId)
                .orElseThrow(() -> new RuntimeException("User not found"));

        // 1. Check if AccountHolder profile exists, if not create it
        AccountHolder holder = holderRepository.findByUser_UserId(userId)
                .orElseGet(() -> {
                    AccountHolder newHolder = new AccountHolder();
                    newHolder.setUser(user);
                    newHolder.setFirstName(request.getFirstName());
                    newHolder.setLastName(request.getLastName());
                    newHolder.setAddress(request.getAddress());
                    newHolder.setPhone(request.getPhone());
                    newHolder.setDob(request.getDob());
                    return holderRepository.save(newHolder);
                });

        // 2. Create the actual Bank Account
        Account account = new Account();
        account.setHolder(holder);
        account.setAccountType(request.getAccountType()); // Savings, Checking
        account.setBalance(BigDecimal.ZERO);
        account.setStatus("Active");
        
        // Generate 10-digit Account Number
        account.setAccountNumber(generateAccountNumber());

        return accountRepository.save(account);
    }

    // --- DEPOSIT LOGIC ---
    @Transactional
    public void deposit(AmountRequest request) {
        Account account = accountRepository.findByAccountNumber(request.getAccountNumber())
                .orElseThrow(() -> new RuntimeException("Account not found"));

        if (request.getAmount().compareTo(BigDecimal.ZERO) <= 0) {
            throw new RuntimeException("Deposit amount must be positive");
        }

        // Update Balance
        account.setBalance(account.getBalance().add(request.getAmount()));
        accountRepository.save(account);

        // Log Transaction
        logTransaction(account, "CREDIT", request.getAmount(), "Cash Deposit");
    }

    // --- WITHDRAW LOGIC ---
    @Transactional
    public void withdraw(AmountRequest request) {
        Account account = accountRepository.findByAccountNumber(request.getAccountNumber())
                .orElseThrow(() -> new RuntimeException("Account not found"));

        if (account.getBalance().compareTo(request.getAmount()) < 0) {
            throw new RuntimeException("Insufficient Funds");
        }

        // Update Balance
        account.setBalance(account.getBalance().subtract(request.getAmount()));
        accountRepository.save(account);

        // Log Transaction
        logTransaction(account, "DEBIT", request.getAmount().negate(), "Cash Withdrawal");
    }

    // Helper to log transactions
    private void logTransaction(Account account, String type, BigDecimal amount, String desc) {
        Transaction tx = new Transaction();
        tx.setAccount(account);
        tx.setTransactionType(type);
        tx.setAmount(amount);
        tx.setDescription(desc);
        tx.setStatus("Completed");
        tx.setTimestamp(LocalDateTime.now());
        transactionRepository.save(tx);
    }

    private String generateAccountNumber() {
        return String.valueOf(1000000000L + new Random().nextLong(9000000000L));
    }
    
    public List<Account> getAccountsByUser(UUID userId) {
         // Assuming you add a derived query in Repo: findByHolder_User_UserId
        return accountRepository.findByHolder_User_UserId(userId);
    }
```

```java
        card.setExpirationDate(LocalDate.now().plusYears(3));

        card.setStatus("Active");
        card.setIssueDate(LocalDate.now());

        return cardRepository.save(card);
```
